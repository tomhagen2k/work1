+    function buildAvReportPlan(iocList) {
+      const plan = {
+        malware: [],
+        connections: [],
+        vulns: [],
+        skipped: [],
+        machines: {}
+      };
+      const publicIp = getOutboundIp();
+
+      iocList.forEach((ioc) => {
+        const type = (ioc.type || "").toUpperCase();
+        if (type === "DOMAIN" || type === "URL") {
+          plan.skipped.push({ id: ioc.id, type, reason: "Không thuộc phạm vi gửi" });
+          return;
+        }
+        const alerts = Array.isArray(ioc.alerts) ? ioc.alerts : [];
+        if (alerts.length === 0 && (type === "SHA256" || type === "MD5" || type === "IPV4" || type === "CVE")) {
+          plan.skipped.push({ id: ioc.id, type, reason: "Chưa gắn cảnh báo với máy" });
+          return;
+        }
+
+        if (type === "SHA256" || type === "MD5") {
+          const ep = resolveEndpoint(alerts[0] || {});
+          plan.malware.push({
+            ip: ep.ip,
+            ippublic: publicIp,
+            name: ep.name,
+            malwareName: ioc.malware_name || "Unknown.Malware",
+            malwareType: ioc.malware_type || "Trojan",
+            malwareBehavior: ioc.malware_behavior || "Unknown",
+            typeOfDevice: "SSD",
+            numberFile: Math.max(1, ioc.hit_count || 1)
+          });
+          plan.machines[ep.name] = ep;
+        } else if (type === "IPV4") {
+          const ep = resolveEndpoint(alerts[0] || {});
+          plan.connections.push({
+            ip: ep.ip,
+            ippublic: publicIp,
+            name: ep.name,
+            program: (alerts[0] && (alerts[0].rule_description || alerts[0].name)) ? "unknown.exe" : "unknown.exe",
+            targetIp: ioc.value
+          });
+          if (alerts[0] && alerts[0].name) {
+            plan.connections[plan.connections.length - 1].program = "svchost_update.exe";
+          }
+          plan.machines[ep.name] = ep;
+        } else if (type === "CVE") {
+          const ep = resolveEndpoint(alerts[0] || {});
+          plan.vulns.push({
+            ip: ep.ip,
+            ippublic: publicIp,
+            name: ep.name,
+            cve: ioc.value,
+            osName: ep.os
+          });
+          plan.machines[ep.name] = ep;
+        } else {
+          plan.skipped.push({ id: ioc.id, type, reason: "Loại dữ liệu không gửi" });
+        }
+      });
+
+      return plan;
+    }
